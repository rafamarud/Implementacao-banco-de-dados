FROM FUNCIONARIO AS F

-- AND --

FROM FUNCIONARIO AS F

WHERE F.Sexo = 'M' and F.Salario >= 30000;



-- OR --
SELECT *
FROM FUNCIONARIO AS F
WHERE F.Endereco LIKE '%São Paulo%' 
	OR F.Endereco LIKE '%Curitiba%';



-- NOT --
SELECT *
FROM FUNCIONARIO AS F
WHERE NOT
	F.Endereco LIKE '%São Paulo%';



-- ORDER BY --

FROM FUNCIONARIO AS F, DEPENDENTE AS D

	(SELECT COUNT(F.Cpf) FROM FUNCIONARIO AS F) + (SELECT COUNT(D.Nome_dependente)FROM DEPENDENTE AS D)
	AS 'QtdPessoas'



--AVG--
SELECT AVG(F.Salario) AS MediaSalarial
FROM FUNCIONARIO AS F

SELECT *
FROM FUNCIONARIO AS F
WHERE F.Salario < (SELECT AVG(F.Salario) AS MediaSalarial FROM FUNCIONARIO AS F)
ORDER BY F.Salario ASC;
